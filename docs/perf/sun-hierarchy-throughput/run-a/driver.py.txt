import subprocess
from pathlib import Path
root = Path.cwd()
for scene, spatial in [("far-128", False), ("far-128", True), ("oblique-far-128", True), ("oblique-far-128", False)]:
    label = "spatial" if spatial else "unsorted"
    output = Path("/tmp/shadow-distinct-qualified-a") / scene / label
    print(f"Measuring {scene}/{label}", flush=True)
    command = ["python3", "scripts/perf/sun_hierarchy_probe.py", f"docs/pr-screenshots/codex/sun-geometry-capture/{scene}/index.csv.gz", "--query-side", "257", "--measure", "--output", str(output)]
    if spatial:
        command.append("--spatial")
    subprocess.run(command, cwd=root, check=True)
