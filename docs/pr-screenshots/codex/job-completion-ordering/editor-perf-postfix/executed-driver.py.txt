"""Capture-free editor shadow cost matrix with an outer quiet-window verdict."""

import argparse
import hashlib
import json
import os
import shutil
import subprocess
import sys
from datetime import datetime, timezone
from pathlib import Path

REPO = Path("/Users/evinjkill/src/IrredenEngine")
sys.path.insert(0, str(REPO / "scripts/perf"))
from compare_perf_runs import parse_report
from profile_matrix import summarize
from repeat_profile import (RENDER_EXPERIMENT_ENV, directory_digest,
                            frame_measurement_failure, host_load, host_power_source)

FRAMES = 360
ROUNDS = 3
BASE = REPO / "build-release/creations/editors/voxel_editor"
BINARY = BASE / "IRVoxelEditor"
CASES = {
    f"yaw{yaw}-zoom{zoom}-surface{surface}-timing{int(timing)}": (yaw, zoom, surface, timing)
    for yaw, zoom in ((30, 4), (67, 8))
    for timing in (True, False)
    for surface in (0, 1)
}


def digest(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def identity():
    return {
        "binary_sha256": digest(BINARY),
        "shader_sha256": directory_digest(BASE / "shaders"),
        "runtime_scripts_sha256": directory_digest(BASE / "scripts"),
        "config_sha256": digest(BASE / "scripts/config.lua"),
    }


def validate_report(data, yaw, zoom, surface, timing):
    failure = frame_measurement_failure(data, FRAMES)
    if failure:
        raise RuntimeError(failure)
    w = data.witness
    if (w.pose_samples != FRAMES or w.yaw_first_deg is None or w.yaw_last_deg is None
            or w.yaw_travel_deg is None or abs(w.yaw_first_deg - yaw) > .01
            or abs(w.yaw_last_deg - yaw) > .01 or abs(w.yaw_travel_deg) > .01
            or w.zoom_min != zoom or w.zoom_max != zoom or w.overflow_max_dropped != 0):
        raise RuntimeError(f"pose/overflow mismatch {w}")
    if (data.gpu_frame.supported is not True or data.gpu_frame.attempted != FRAMES
            or data.gpu_frame.valid != FRAMES or data.gpu_frame.invalid != 0):
        raise RuntimeError(f"incomplete GPU frame coverage {data.gpu_frame}")
    sampled_stages = [s for s in data.gpu_stages if (s.samples or 0) > 0]
    if not timing:
        if sampled_stages:
            raise RuntimeError("instrumentation-off control contains GPU stage samples")
        return
    wanted = "perAxisSurfaceFb" if surface else "lightingPerAxis"
    forbidden = "lightingPerAxis" if surface else "perAxisSurfaceFb"
    stage = data.gpu_by_name(wanted)
    other = data.gpu_by_name(forbidden)
    if stage is None or not stage.samples or (other is not None and other.samples):
        raise RuntimeError(f"missing or contradictory GPU route witness: {wanted}")


def measure(root):
    if os.environ.get("IR_ACQUIRE_HELD_VERB") != "benchmark":
        raise RuntimeError("measurements require the outer ir-acquire benchmark lease")
    expected = identity()
    power = host_power_source()
    (root / "cases.json").write_text(json.dumps({
        "cases": CASES, "rounds": ROUNDS, "frames": FRAMES,
        "order": "forward on odd rounds; reverse on even rounds",
        "window_mode": "offscreen", "artifact_identity": expected, "power": power,
        "instrumentation_control": "GPU stage timestamps off; GPU frame envelope retained",
    }, indent=2) + "\n")
    for round_index in range(1, ROUNDS + 1):
        order = list(CASES) if round_index % 2 else list(reversed(CASES))
        for name in order:
            yaw, zoom, surface, timing = CASES[name]
            out = root / name / f"round-{round_index}"
            out.mkdir(parents=True)
            cmd = ["fleet-run", "--release", "--timeout", "90", "IRVoxelEditor",
                   "--profile-frames", str(FRAMES), "--profile-yaw", str(yaw),
                   "--profile-zoom", str(zoom), "--window-mode", "offscreen"]
            if not timing:
                cmd.append("--no-stage-timing")
            env = dict(os.environ, IR_RUN_NO_DISPLAY_GRACE="30",
                       IR_PERAXIS_SURFACE_LIGHTING=str(surface),
                       IR_PERAXIS_VISIBILITY_PREPASS="0", IR_PERAXIS_VISIBILITY_STATS="0")
            report = BASE / "save_files/profile_report.txt"
            old = report.stat().st_mtime_ns if report.exists() else None
            started = datetime.now(timezone.utc).isoformat()
            before_load = host_load()
            with (out / "run.log").open("w") as stream:
                result = subprocess.run(cmd, env=env, cwd=REPO, stdout=stream,
                                        stderr=subprocess.STDOUT)
            if result.returncode or "RESULT=CLEAN" not in (out / "run.log").read_text():
                raise RuntimeError(f"{name} failed exit {result.returncode}")
            if not report.exists() or report.stat().st_mtime_ns == old:
                raise RuntimeError("missing or stale report")
            shutil.copy2(report, out / "run-1.txt")
            data = parse_report(out / "run-1.txt", name)
            validate_report(data, yaw, zoom, surface, timing)
            if identity() != expected or host_power_source() != power:
                raise RuntimeError("runtime artifacts or power changed")
            (out / "manifest.json").write_text(json.dumps({
                "command": cmd, "surface": surface, "stage_timing": timing,
                **expected, "host_power": power, "host_load_before": before_load,
                "host_load_after": host_load(), "started": started,
                "finished": datetime.now(timezone.utc).isoformat(),
                "render_environment": {key: env.get(key) for key in RENDER_EXPERIMENT_ENV},
                "witness": vars(data.witness), "report_sha256": digest(out / "run-1.txt"),
            }, indent=2) + "\n")
            print(name, round_index, "steady", data.steady_frame.avg, "gpu",
                  [(s.name, s.avg_ms) for s in data.gpu_frame.metrics], flush=True)
        summarize(root, CASES)
    (root / "measurement-complete.json").write_text(json.dumps({
        "cases": len(CASES), "rounds": ROUNDS, "runs": len(CASES) * ROUNDS,
    }, indent=2) + "\n")


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--output", type=Path,
                        default=Path("/tmp/irreden-yaw-lighting/editor-perf"))
    parser.add_argument("--inside-benchmark", action="store_true", help=argparse.SUPPRESS)
    args = parser.parse_args()
    root = args.output.resolve()
    if args.inside_benchmark:
        measure(root)
        return 0
    root.mkdir(parents=True, exist_ok=False)
    shutil.copy2(__file__, root / "executed-driver.py.txt")
    quiet = root / "quiet.txt"
    command = ["ir-acquire", "benchmark", "--", sys.executable, str(Path(__file__).resolve()),
               "--output", str(root), "--inside-benchmark"]
    env = dict(os.environ, IR_QUIET_REPORT_FILE=str(quiet))
    with (root / "outer.log").open("w") as stream:
        result = subprocess.run(command, env=env, cwd=REPO, stdout=stream,
                                stderr=subprocess.STDOUT)
    verdict = quiet.read_text().strip() if quiet.exists() else "MISSING"
    qualified = (result.returncode == 0 and verdict == "GUARDED"
                 and (root / "measurement-complete.json").exists())
    (root / "qualification.json").write_text(json.dumps({
        "command": command, "wrapper_exit": result.returncode,
        "quiet_verdict": verdict, "qualified": qualified,
    }, indent=2) + "\n")
    print(root, "GUARDED" if qualified else "UNQUALIFIED", flush=True)
    return 0 if qualified else 1


if __name__ == "__main__":
    raise SystemExit(main())
